# Quick Actions Feature Implementation Plan
**Task: IT-468 — Quick Actions management in Board Settings**
**Requested by: @avoronkov**

## Executive Summary

This plan describes the implementation of a **Quick Actions** feature that allows board admins to define action buttons displayed on cards. Buttons have visibility conditions and execute one or more actions (set/clear properties, add comments) when clicked.

The implementation will follow existing patterns in the codebase, particularly:
- Board Settings UI structure (similar to PropertiesSection, StatusTransitionSection)
- Filter/condition builder UI (filterEntry, filterValue components)
- Property update patterns (mutator.changePropertyValue)
- Card detail view structure (insertion between properties and GitHub blocks)

```go
// Based on: server/model/board.go IsValid()

func (qa *QuickAction) IsValid() error {
    if qa.ID == "" {
        return fmt.Errorf("quick action ID is required")
    }
    if qa.Name == "" {
        return fmt.Errorf("quick action name is required")
    }
    if qa.ConfirmRequired && qa.ConfirmText == "" {
        return fmt.Errorf("confirmation text is required when confirmation is enabled")
    }
    if len(qa.Actions) == 0 {
        return fmt.Errorf("quick action must have at least one action")
    }
    // Validate conditions
    for _, cond := range qa.Conditions {
        if cond.PropertyID == "" {
            return fmt.Errorf("condition propertyId is required")
        }
        if !isValidOperator(cond.Operator) {
            return fmt.Errorf("invalid operator: %s", cond.Operator)
        }
    }
    // Validate actions
    for _, action := range qa.Actions {
        if err := action.IsValid(); err != nil {
            return err
        }
    }
    return nil
}
```

### 3.3 Action Execution Handler

**File:** `server/api/quickActions.go` (NEW)

```go
// POST /boards/{boardID}/cards/{cardID}/quickActions/{actionID}
func (a *API) handleExecuteQuickAction(w http.ResponseWriter, r *http.Request) {
    boardID := mux.Vars(r)["boardID"]
    cardID := mux.Vars(r)["cardID"]
    actionID := mux.Vars(r)["actionID"]
    userID := getUserID(r)

    // Get board
    board, err := a.app.GetBoard(boardID)
    if err != nil {
        a.errorResponse(w, r, err)
        return
    }

    // Check permissions
    if !a.permissions.HasPermissionToBoard(userID, boardID, model.PermissionManageBoardCards) {
        a.errorResponse(w, r, model.NewErrPermission("access denied"))
        return
    }

    // Get quick action
    quickActions := getQuickActionsFromBoard(board)
    var action *model.QuickAction
    for _, qa := range quickActions {
        if qa.ID == actionID {
            action = &qa
            break
        }
    }
    if action == nil {
        a.errorResponse(w, r, model.NewErrNotFound("quick action"))
        return
    }

    // Get card
    card, err := a.app.GetBlock(cardID)
    if err != nil {
        a.errorResponse(w, r, err)
        return
    }

    // Execute actions in a transaction
    err = a.app.ExecuteQuickAction(board, card, action, userID)
    if err != nil {
        a.errorResponse(w, r, err)
        return
    }

    a.updateCardResponse(w, r, userID, boardID, cardID)
}
```

### 3.4 Action Execution Logic

**File:** `server/app/quickActions.go` (NEW)

```go
// Based on: server/app/block.go

func (a *App) ExecuteQuickAction(board *model.Board, card *model.Block, action *model.QuickAction, userID string) error {
    // Execute each action in order
    for _, actionDef := range action.Actions {
        switch actionDef.Type {
        case model.QuickActionSetProperty:
            value := actionDef.Value
            // Handle special values
            if value == "{current_user}" {
                value = userID
            } else if value == "{now}" {
                value = fmt.Sprintf("%d", time.Now().UnixMilli())
            }
            if err := a.SetCardProperty(card, actionDef.PropertyID, value); err != nil {
                return err
            }

        case model.QuickActionClearProperty:
            if err := a.ClearCardProperty(card, actionDef.PropertyID); err != nil {
                return err
            }

        case model.QuickActionAddComment:
            if err := a.AddCardComment(card, actionDef.Text, userID); err != nil {
                return err
            }
        }
    }
    return nil
}
```

```typescript
// Handles special values: {current_user}, {now}

export function evaluateConditions(
    conditions: QuickActionCondition[],
    card: Card,
    board: Board,
    currentUserId: string
): boolean {
    // ALL conditions must be true (AND logic)
    return conditions.every((condition) => {
        const propertyTemplate = board.cardProperties.find(p => p.id === condition.propertyId)
        if (!propertyTemplate) return false

        const cardValue = card.fields.properties[condition.propertyId]

        // Handle special values
        const resolvedValues = resolveSpecialValues(condition.values || [], currentUserId)

        return evaluateCondition(condition.operator, cardValue, resolvedValues, propertyTemplate)
    })
}

function resolveSpecialValues(values: string[], currentUserId: string): string[] {
    return values.map(v => {
        if (v === '{current_user}') return currentUserId
        if (v === '{now}') return String(Date.now())
        return v
    })
```

```typescript
async executeQuickAction(boardId: string, cardId: string, actionId: string): Promise<void> {
    const url = `${this.getBaseURL()}/boards/${boardId}/cards/${cardId}/quickActions/${actionId}`
    const options = {
        method: 'POST',
        headers: {'X-Requested-With': 'XMLHttpRequest'}
    }
    const response = await fetch(url, options)
    if (!response.ok) {
        throw new Error(`Failed to execute quick action: ${response.statusText}`)

}
```

## Potential Pitfalls & Edge Cases

### 1. Special Value Resolution
- **Issue:** `{current_user}` and `{now}` must be resolved at execution time, not at configuration time
- **Solution:** Resolve on backend when executing actions

### 2. Property Type Changes
- **Issue:** If a property used in a Quick Action condition is deleted or type-changed
- **Solution:**
  - Show warning in UI if property no longer exists
  - Evaluate conditions safely; treat missing properties as "not matching"

### 3. Empty vs Not Empty Operators
- **Issue:** These operators should NOT require a value input
- **Solution:** Hide value input when `empty` or `not empty` selected

### 4. Multi-value Properties
- **Issue:** `multiSelect` and `multiPerson` properties store arrays
- **Solution:** Condition evaluator must handle both string and string[] values

### 5. Date Comparisons
- **Issue:** Dates stored as timestamps, need numeric comparison
- **Solution:** Convert to numbers before `>`, `<`, `>=`, `<=` comparisons

### 6. Checkbox Properties
- **Issue:** Only `checked`/`not checked` operators are valid
- **Solution:** Restrict operator options based on property type

### 7. System Properties (createdBy, createdTime, etc.)
- **Issue:** These are read-only but valid for conditions
- **Solution:** Include in property selector but mark as read-only

### 8. Confirmation Dialog Validation
- **Issue:** If `confirmRequired` is true, `confirmText` must not be empty
- **Solution:** Validation in both frontend and backend

### 9. Action Execution Errors
- **Issue:** Partial execution if one action fails
- **Solution:** Execute all actions in a transaction; rollback on failure

### 10. Circular Dependencies
- **Issue:** Quick Action setting a property that would make itself no longer visible
- **Solution:** Accept this behavior; buttons don't need to persist after clicking

### 11. Permission Bypass
- **Issue:** Quick Action should not allow users to do things they couldn't do otherwise
- **Solution:** Check `PermissionManageBoardCards` before executing; property changes are already permission-controlled

### 12. Deleted Properties
- **Issue:** Quick Actions referencing deleted properties
- **Solution:**
  - UI shows warning
  - Evaluation treats as non-matching condition
  - Action execution fails gracefully if property doesn't exist
